```move
use atomica::eth_proof::{Self, StateProof};

/// Create auction with Ethereum lock proof
/// Verifies that seller has locked sufficient FAKETH on Ethereum
public entry fun create_auction_with_proof(
    seller: &signer,
    proof_block_number: u64,
    proof_block_hash: vector<u8>,
    proof_state_root: vector<u8>,
    proof_contract_address: vector<u8>,
    proof_user_address: vector<u8>,
    proof_token_address: vector<u8>,
    proof_storage_key: vector<u8>,
    proof_storage_value: u256,
    proof_account_proof: vector<vector<u8>>,
    proof_storage_proof: vector<vector<u8>>,
    amount: u64,
    min_price: u64,
    duration: u64,
    mpk: vector<u8>
) {
    // 1. Create proof struct
    let proof = eth_proof::create_proof(
        proof_block_number,
        proof_block_hash,
        proof_state_root,
        proof_contract_address,
        proof_user_address,
        proof_token_address,
        proof_storage_key,
        proof_storage_value,
        proof_account_proof,
        proof_storage_proof
    );
    
    // 2. Verify proof and check locked amount
    assert!(
        eth_proof::has_sufficient_lock(&proof, (amount as u256)),
        E_INSUFFICIENT_LOCKED
    );
    
    // 3. Create auction (existing logic)
    create_auction(seller, amount, min_price, duration, mpk);
}

const E_INSUFFICIENT_LOCKED: u64 = 100;
```

### Step 5: End-to-End Integration Test

**File to Create:** `source/atomica-web/tests/e2e/cross-chain-auction.test.ts`

```typescript
import { describe, it, expect } from 'vitest';
import { EthereumDockerTestnet } from '@atomica/ethereum-docker-testnet';
import { DockerTestnet } from '@atomica/docker-testnet';
import { generateLockedBalanceProof } from '../../src/lib/ethereum/proofs/generator';
import { Aptos } from '@aptos-labs/ts-sdk';

describe('Cross-Chain Auction Flow', () => {
  it('should create auction after locking FAKETH on Ethereum', async () => {
    // 1. Start both testnets
    const [ethTestnet, aptosTestnet] = await Promise.all([
      EthereumDockerTestnet.start(8),
      DockerTestnet.new(4)
    ]);
    
    // 2. Deploy contracts
    const lockBox = await deployLockBox(ethTestnet);
    const registry = await deployAptosContracts(aptosTestnet);
    
    // 3. Seller locks 10 FAKETH
    const seller = await ethTestnet.getAccount(0);
    await lockBox.lock(fakeETHAddress, ethers.parseEther("10"));
    
    // 4. Wait for block finality
    await ethTestnet.waitForBlocks(2);
    
    // 5. Generate proof
    const proof = await generateLockedBalanceProof(
      ethTestnet.rpcUrl,
      lockBox.address,
      seller.address,
      fakeETHAddress,
      await ethTestnet.getBlockNumber() - 1
    );
    
    // 6. Submit proof to Aptos and create auction
    const aptos = new Aptos(aptosTestnet.aptosConfig);
    const tx = await aptos.transaction.build.simple({
      sender: sellerAptosAddress,
      data: {
        function: `${registry}::registry::create_auction_with_proof`,
        functionArguments: [
          proof.blockNumber,
          proof.blockHash,
          proof.stateRoot,
          // ... all proof fields ...
          10, // 10 FAKETH
          1000, // min 1000 FAKEUSD
          3600, // 1 hour
          mpk
        ]
      }
    });
    
    const result = await aptos.signAndSubmitTransaction({
      signer: sellerAptosSigner,
      transaction: tx
    });
    
    await aptos.waitForTransaction({ transactionHash: result.hash });
    
    // 7. Verify auction was created
    const auctions = await registry.getAuctions();
    expect(auctions.length).toBe(1);
    expect(auctions[0].amount).toBe(10);
    
    // 8. Cleanup
    await Promise.all([
      ethTestnet.stop(),
      aptosTestnet.stop()
    ]);
  }, 300000); // 5 minute timeout
});
```

## Important Notes

1. **Block Finality:** Always use `blockNumber - 1` or older for proof generation to ensure block won't be reorged

2. **Storage Key:** Must exactly match Solidity's `calculateStorageKey()` result

3. **Proof Size:** Ethereum proofs can be large (5-10 KB). Ensure Aptos transaction size limits aren't exceeded.

4. **Gas Costs:** MPT verification is expensive (~2000-5000 gas on Aptos). Consider proof caching for production.

5. **Error Handling:** Provide clear error messages:
   - "Insufficient locked amount"
   - "Invalid proof"
   - "Proof too old (block finality)"
